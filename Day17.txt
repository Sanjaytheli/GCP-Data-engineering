Retail_Summary Flow: (Retail_Summary.sql)
--------------

create database if not exists processed_db
location 'gs://cdebatch5758sparkusecase/processed.db';

use processed_db;

create table if not exists processed_db.Retail_Summary
(
name string,
city string,
Tot_Purchase_Amnt decimal(10,2)
)using PARQUET;

insert into processed_db.Retail_Summary
select sales.name,cust.city,coalesce(sum(ord.purch_amt),0) as Tot_Purchase_Amnt from
(select * from raw_db.customer_lo union select * from raw_db.customer_ny)cust
join (select * from raw_db.orders)ord
on cust.customer_id=ord.customer_id
join (select * from raw_db.salesman_lo union select * from raw_db.salesman_ny)sales
on sales.salesman_id=ord.salesman_id
group by sales.name,cust.city;

customer_lo flow:  {Parameterized approach} {common.sql}
------------
create database if not exists raw_db
location 'gs://${bucket_name}/raw_db.db';

use raw_db;

create or replace temporary view ${table_name}_v
using JSON
options (
path='gs://${bucket_name}/source/${table_name}.json'
);

create table if not exists raw_db.${table_name}
using PARQUET
select * from ${table_name}_v;
